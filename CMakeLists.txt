cmake_minimum_required(VERSION 3.24)
project(DiskScope VERSION 0.1.3 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(DS_GUI "Build the GTK explorer" ON)
option(DS_SANITIZE "Enable ASan and UBSan for project code" OFF)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options("-ffile-prefix-map=${CMAKE_CURRENT_SOURCE_DIR}=."
                        "-ffile-prefix-map=${CMAKE_CURRENT_BINARY_DIR}=./build")
endif()
include(CTest)
include(cmake/Dependencies.cmake)

function(ds_c_target target)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wformat=2)
        if(DS_SANITIZE AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined -no-pie)
        endif()
    endif()
endfunction()

add_library(ds_core STATIC src/core/metrics.c src/core/path.c src/core/policy.c)
if(WIN32)
    target_sources(ds_core PRIVATE src/core/win_io.c)
endif()
target_include_directories(ds_core PUBLIC include)
target_link_libraries(ds_core PUBLIC PkgConfig::GLIB)
ds_c_target(ds_core)
if(DS_SANITIZE AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_options(ds_core INTERFACE -fsanitize=address,undefined -no-pie)
endif()

function(ds_test name)
    string(REPLACE "." "_" target "test_${name}")
    add_executable(${target} ${ARGN})
    target_link_libraries(${target} PRIVATE ds_core)
    ds_c_target(${target})
    add_test(NAME ${name} COMMAND ${target})
endfunction()

if(BUILD_TESTING)
    foreach(unit metrics path policy)
        ds_test(unit.${unit} tests/unit/test_${unit}.c)
    endforeach()
endif()
